#
# WorldGuard 黑名单
#
# 黑名单可以让你阻止某些动作、方块和物品被使用。
# 你选择一组"受影响的物品"和一组"要执行的动作"。
#
###############################################################################
#
# 阻止挖矿和放置部分矿石的示例：
# [coal_ore,gold_ore,iron_ore]
# on-break=deny,log,kick
# on-place=deny,tell
#
# 你可以检测的事件：
# - on-break（该类型的方块即将被破坏时）
# - on-destroy-with（玩家破坏时所持有的物品/方块）
# - on-place（方块正在被放置时）
# - on-use（正在使用物品，如打火石或水桶时）
# - on-interact（方块被使用时（门、箱子等））
# - on-drop（物品从玩家背包中丢弃时）
# - on-acquire（物品通过某种方式进入玩家背包时）
# - on-equip（物品装备到玩家的盔甲栏时）
#      注意：由于 Bukkit API 的缺陷，on-equip 过于严格
# - on-dispense（发射器即将发射物品时）
#
# 动作（用于事件）：
# - deny（完全拒绝，用于黑名单模式）
# - allow（用于白名单模式）
# - notify（通知拥有 'worldguard.notify' 权限的管理员）
# - log（记录到控制台/文件/数据库）
# - tell（告知玩家这是不允许的）
# - kick（踢出玩家）
# - ban（封禁玩家）
#
# 选项：
# - ignore-groups（以逗号分隔、不受影响的群组列表）
# - ignore-perms（以逗号分隔、不受影响的权限列表 - 可以自定义！）
# - comment（用于 'log' 和 'notify' 的备注信息）
# - message（向用户显示的可选消息；%s 是物品名称）
#
###############################################################################
#
# 更多信息，请参阅：
# https://worldguard.enginehub.org/en/latest/blacklist/
#
###############################################################################
#
# 下面是一些示例。
# 请记住：如果某行前面有 #，它将被忽略。
#

# Deny lava buckets
#[lava_bucket]
#ignore-perms=my.own.madeup.permission
#ignore-groups=admins,mods
#on-use=deny,tell

# Deny some ore
#[coal_ore,gold_ore,iron_ore]
#ignore-groups=admins,mods
